cmake_minimum_required(VERSION 3.13)

# initialize pico-sdk from submodule
# note: this must happen before project()
include(pico-sdk/pico_sdk_init.cmake)

project(roadrunner)

set(ROADRUNNER_FIRMWARE_VERSION "dev" CACHE STRING
    "Roadrunner firmware version included in USB admin INFO responses")

# initialize the Raspberry Pi Pico SDK
pico_sdk_init()

file(MAKE_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}/generated)

# UART with RGB LED ordering
add_executable(roadrunner_v1_uart_rgb)
pico_enable_stdio_usb(roadrunner_v1_uart_rgb 1)
pico_enable_stdio_uart(roadrunner_v1_uart_rgb 0)
pico_generate_pio_header(roadrunner_v1_uart_rgb ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio OUTPUT_DIR ${CMAKE_CURRENT_LIST_DIR}/generated)
target_sources(roadrunner_v1_uart_rgb PRIVATE neopixel.c as5600.c ir_sensor.c tmcuart.c main.c identity_record.c identity_registers.c identity_store.c usb_admin.c usb_descriptor_strings.c usb_descriptors.c)
target_include_directories(roadrunner_v1_uart_rgb PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(roadrunner_v1_uart_rgb PRIVATE pico_stdlib hardware_pio hardware_i2c hardware_flash pico_flash tinyusb_device)
target_compile_definitions(roadrunner_v1_uart_rgb PRIVATE PICO_FLASH_ASSUME_CORE1_SAFE=1 PICO_STDIO_USB_USE_DEFAULT_DESCRIPTORS=0 PICO_STDIO_USB_ENABLE_TINYUSB_INIT=0 PICO_STDIO_USB_ENABLE_IRQ_BACKGROUND_TASK=0 ROADRUNNER_FIRMWARE_VERSION=\"${ROADRUNNER_FIRMWARE_VERSION}\")
pico_add_linker_script_override_path(roadrunner_v1_uart_rgb ${CMAKE_CURRENT_LIST_DIR}/cmake FILES pico_flash_region.ld)
pico_add_extra_outputs(roadrunner_v1_uart_rgb)

# UART with GRB LED ordering
